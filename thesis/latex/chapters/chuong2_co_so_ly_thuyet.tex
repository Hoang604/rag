\section{Khảo sát các công trình liên quan}
\label{sec:c2_lien_quan}

\subsection{Truy hồi và hỏi đáp văn bản pháp luật tiếng Việt}

\textbf{Các cuộc thi và bộ dữ liệu.} Zalo AI Challenge 2021 có nhánh Legal Text Retrieval \cite{zalo2021legal}: với mỗi câu hỏi pháp luật tiếng Việt, hệ thống cần tìm các điều luật liên quan; bản dữ liệu được dùng rộng rãi là ``ZacLegalTextRetrieval'' trên Hugging Face và MTEB \cite{greennode_zalolegal}, với khoảng 61 nghìn đoạn. ALQAC 2021--2023 \cite{nguyen2021alqac,nguyen2022alqac,nguyen2023alqac} tổ chức cùng hội nghị KSE gồm các nhiệm vụ tìm điều luật và hỏi đáp trên luật Việt Nam.

Các bộ dữ liệu gần đây mở rộng sang hỏi đáp và suy luận: VLQA \cite{nguyen2025vlqa} đánh giá cả tìm kiếm lẫn hỏi đáp; VLegal-Bench \cite{dong2025vlegalbench} gồm 10.450 mẫu theo các mức độ nhận thức; ViBidLQA \cite{nguyen2024vibidlqa} so sánh hướng trích xuất (MRC) với hướng sinh câu trả lời trên luật đấu thầu. Gần với đề tài nhất là VLSP 2025 MLQA-TSR \cite{luu2025vlsp}, hỏi đáp đa phương thức về biển báo giao thông đường bộ, với kết quả tốt nhất F2 = 64,55\% ở nhiệm vụ tìm điều luật.

\textbf{Các phương pháp.} Pham và cộng sự \cite{pham2022multistage} kết hợp điểm BM25 với điểm cosine của sentence-transformers trên nền PhoBERT \cite{nguyen2020phobert}, đạt F2 = 0,741 trên kho khoảng 114 nghìn điều luật. Nguyen và cộng sự \cite{nguyen2024attentive} đề xuất các mạng chú ý phân cấp cho tìm văn bản pháp luật đa ngôn ngữ. Nhóm BKAI \cite{duc2024bkai} huấn luyện bi-encoder trên PhoBERT-base-v2, đạt Acc@1 = 73,28 trên Zalo 2021. Pham Tien và cộng sự \cite{pham2024synthetic} tiền huấn luyện bi-encoder và ColBERT bằng câu hỏi do mô hình ngôn ngữ lớn sinh ra. Nguyen Ba và cộng sự \cite{nguyenba2024legalir} hợp nhất tìm theo từ khoá và theo vector bằng một biến thể của RRF rồi xếp hạng lại bằng mô hình ngôn ngữ lớn, gần với thiết kế của đề tài nhất. Phần lớn các công trình này tách từ bằng VnCoreNLP \cite{vu2018vncorenlp} hoặc pyvi \cite{tran_pyvi} trước khi mã hoá.

\subsection{Sinh câu trả lời có tăng cường truy hồi và agent}

Lewis và cộng sự \cite{lewis2020rag} đề xuất mô hình RAG, kết hợp mô hình sinh với bộ tìm kiếm vector DPR \cite{karpukhin2020dpr} để tri thức nằm một phần trong chỉ mục ngoài có thể thay thế. Gao và cộng sự \cite{gao2023ragsurvey} chia RAG thành ba nhóm: \emph{Naive RAG} tìm một lần rồi sinh; \emph{Advanced RAG} thêm viết lại câu hỏi và xếp hạng lại; \emph{Modular RAG} ghép linh hoạt các thành phần và cho phép lặp. Về đơn vị tìm kiếm, Jimeno Yepes và cộng sự \cite{jimeno2024chunking} cắt theo loại phần tử thay vì độ dài cố định, Chen và cộng sự \cite{chen2023densex} cho thấy đơn vị mệnh đề tốt hơn đoạn văn, còn Anthropic \cite{anthropic2024contextual} bổ sung ngữ cảnh vào đầu mỗi đoạn và giảm tỉ lệ tìm trượt từ 5,7\% xuống 2,9\%. LegalBench-RAG \cite{pipitone2024legalbenchrag} là bộ đánh giá bước tìm kiếm của RAG trong miền pháp lý tiếng Anh.

ReAct \cite{yao2023react} cho mô hình ngôn ngữ xen kẽ suy luận với hành động như gọi API tìm kiếm. Toolformer \cite{schick2023toolformer} và Self-RAG \cite{asai2024selfrag} huấn luyện mô hình tự quyết định khi nào gọi công cụ hoặc tìm kiếm; IRCoT \cite{trivedi2023ircot} dùng từng bước suy luận làm truy vấn cho lần tìm kế tiếp. Singh và cộng sự \cite{singh2025agentic} tổng hợp các hệ thống RAG dạng agent, và Model Context Protocol \cite{anthropic2024mcp} chuẩn hoá cách ứng dụng mô hình ngôn ngữ kết nối với công cụ bên ngoài.

\subsection{Văn bản quy phạm pháp luật và cấu trúc phân cấp}
\label{subsec:c2_vbqppl}

Văn bản quy phạm pháp luật Việt Nam được sắp theo thứ bậc của cơ quan ban hành: luật quy định nguyên tắc chung; nghị định quy định chi tiết và xử phạt; thông tư hướng dẫn kỹ thuật và thủ tục; quy chuẩn kỹ thuật ban hành kèm theo thông tư. Theo quy ước soạn thảo, nội dung được chia thành các cấp lồng nhau Phần, Chương, Mục, Tiểu mục, Điều, Khoản, Điểm với cách đánh số riêng: Chương dùng số La Mã, Điều và Khoản dùng số Ả Rập, Điểm dùng chữ cái kèm ngoặc đơn (``a)'', ``đ)''); các cấp chèn thêm khi sửa đổi mang hậu tố như \texttt{12a}, \texttt{a1} (Hình~\ref{fig:c2_cay_van_ban}).

Hai quy ước có ý nghĩa đặc biệt với truy hồi. Thứ nhất, mức tiền nằm ở câu dẫn của Khoản còn hành vi nằm ở các Điểm, nên một Điểm đứng riêng không đủ để xác định mức phạt. Thứ hai, văn bản sửa đổi chép lời văn mới trong ngoặc kép (``a) Sửa đổi, bổ sung điểm đ như sau:'' rồi đến dòng ``đ)''), nên nếu không xử lý ngoặc kép, dòng này bị hiểu nhầm là một Điểm của chính văn bản sửa đổi. Ngoài ra, trích dẫn nội bộ theo mẫu ổn định như ``điểm a khoản 9 Điều này'' cho phép liên kết trích dẫn một cách tất định.

\subsection{Mô hình BM25}
\label{subsec:c2_bm25}

BM25 \cite{robertson2009bm25} là hàm xếp hạng theo từ khoá kết hợp trọng số IDF (từ hiếm quan trọng hơn từ phổ biến), tần suất từ có bão hoà (tham số $k_1$) và chuẩn hoá theo độ dài (tham số $b$). Với kho gồm $N$ đoạn và $n_t$ là số đoạn chứa từ $t$:
\begin{equation}
\label{eq:c2_idf}
\mathrm{IDF}(t) = \ln\left(1 + \frac{N - n_t + 0{,}5}{n_t + 0{,}5}\right).
\end{equation}
\begin{equation}
\label{eq:c2_bm25}
\mathrm{BM25}(q, d) = \sum_{t \in q} \mathrm{IDF}(t) \cdot \frac{\mathrm{tf}(t, d)\,(k_1 + 1)}{\mathrm{tf}(t, d) + k_1 \left(1 - b + b\,\dfrac{|d|}{\mathrm{avgdl}}\right)},
\end{equation}
trong đó $\mathrm{tf}(t, d)$ là số lần từ $t$ xuất hiện trong $d$, $|d|$ là độ dài của $d$ và $\mathrm{avgdl}$ là độ dài trung bình của các đoạn; hệ thống dùng giá trị thông dụng $k_1 = 1{,}2$, $b = 0{,}75$. Hàm \texttt{ts\_rank} có sẵn của PostgreSQL không có IDF, nên những từ xuất hiện ở hầu hết các đoạn như ``xe'' hay ``giao thông'' nặng ngang từ hiếm, và cũng không chuẩn hoá theo độ dài.

Đề tài áp dụng BM25 cho tiếng Việt mà không tách từ: văn bản được bỏ dấu, chuyển chữ thường và tách thành âm tiết đơn cùng cặp hai âm tiết liền kề (``đèn tín hiệu'' sinh \texttt{den\_tin} và \texttt{tin\_hieu}). Vì cả kho và câu hỏi đều được bỏ dấu, câu gõ không dấu vẫn khớp được với văn bản có dấu.

\subsection{Tìm kiếm theo vector và độ tương tự cosine}
\label{subsec:c2_dense}

Tìm kiếm theo vector (dense retrieval) mã hoá câu hỏi và đoạn văn vào cùng một không gian sao cho các cặp liên quan nằm gần nhau. DPR \cite{karpukhin2020dpr} vượt BM25 từ 9 đến 19 điểm phần trăm về độ chính xác top-20 trên các bộ hỏi đáp miền mở, nhưng BEIR \cite{thakur2021beir} cho thấy BM25 vẫn là mốc so sánh vững và nhiều mô hình vector giảm chất lượng khi chuyển sang miền mới. Mức gần nhau được đo bằng độ tương tự cosine:

\begin{equation}
\label{eq:c2_cosine}
\cos(\mathbf{q}, \mathbf{d}) = \frac{\mathbf{q} \cdot \mathbf{d}}{\lVert \mathbf{q} \rVert \, \lVert \mathbf{d} \rVert}.
\end{equation}
Hệ thống dùng chỉ mục HNSW, một đồ thị lân cận nhiều tầng, để tìm lân cận gần đúng mà không phải so câu hỏi với mọi đoạn. Theo Matryoshka Representation Learning \cite{kusupati2022mrl}, các tiền tố của vector embedding cũng là biểu diễn dùng được, nên hệ thống giữ 512 trên 1024 chiều đầu rồi chuẩn hoá lại theo L2, giảm một nửa dung lượng lưu trữ và chi phí tính khoảng cách.

\begin{equation}
\label{eq:c2_mrl}
\tilde{\mathbf{v}} = \frac{(v_1, v_2, \ldots, v_{512})}{\lVert (v_1, v_2, \ldots, v_{512}) \rVert_2}.
\end{equation}

\subsection{Reciprocal Rank Fusion}
\label{subsec:c2_rrf}

Do nhánh vector và nhánh BM25 có thang điểm khác nhau, Reciprocal Rank Fusion \cite{cormack2009rrf} hợp nhất hai danh sách chỉ dựa trên thứ hạng:
\begin{equation}
\label{eq:c2_rrf}
\mathrm{RRF}(d) = \sum_{r \in R} \frac{1}{k + r(d)},
\end{equation}
trong đó $R$ là tập các danh sách xếp hạng và $r(d)$ là thứ hạng của $d$ trong danh sách $r$. Bài báo gốc dùng $k = 60$; Bruch và cộng sự \cite{bruch2023fusion} chỉ ra RRF nhạy với tham số này. Với $k = 60$, hạng 1 ($1/61 \approx 0{,}0164$) và hạng 10 ($1/70 \approx 0{,}0143$) gần như ngang nhau, nên một kết quả đứng đầu ở một nhánh có thể bị một kết quả đứng thứ mười ở cả hai nhánh vượt qua. Đề tài chọn $k = 20$, khi đó hạng 1 nhận điểm gấp khoảng 1,6 lần hạng 10, giúp giữ lại kết quả rất tốt của từng nhánh; chênh lệch so với $k = 60$ trên tập dev chỉ là 4 câu.

\subsection{Mô hình embedding Qwen3-Embedding}
\label{subsec:c2_qwen3}

Qwen3 Embedding \cite{zhang2025qwen3} là họ mô hình embedding dựng trên Qwen3, hỗ trợ chọn số chiều theo kiểu Matryoshka và nhận chỉ dẫn cho câu hỏi. Bản 0.6B (1024 chiều) đạt 64,33 điểm MMTEB \cite{enevoldsen2025mmteb}, cao hơn multilingual-e5-large-instruct \cite{wang2024me5} (63,22) và BGE-M3 \cite{chen2024m3} (59,56); bản 4B và 8B đạt 69,45 và 70,58.

Hệ thống dùng bản 0.6B vì đa ngôn ngữ và chạy được trên GPU 4 GB của máy phát triển; bản 4B cần khoảng 8 GB nên chưa được thử. Câu hỏi được mã hoá kèm prompt truy vấn (\texttt{Instruct: \ldots{} Query: \ldots}), còn đoạn văn bản được mã hoá không kèm prompt. Vì MTEB \cite{muennighoff2023mteb} và MMTEB cho thấy không có mô hình nào thắng ở mọi nhiệm vụ, việc đánh giá trực tiếp trên dữ liệu của bài toán là cần thiết.

\subsection{Agent dựa trên mô hình ngôn ngữ lớn và giao thức MCP}
\label{subsec:c2_agent_mcp}

Trong Naive RAG \cite{gao2023ragsurvey}, câu hỏi được dùng trực tiếp làm truy vấn và mô hình trả lời trong một lượt, nên nếu cách diễn đạt của câu hỏi khiến bộ tìm kiếm lệch chủ đề thì mô hình không có cơ hội khắc phục. Agent theo kiểu ReAct \cite{yao2023react} lặp lại các bước suy luận, gọi công cụ, đọc kết quả, rồi quyết định tìm lại, đọc thêm hay trả lời (Hình~\ref{fig:c2_rag_agent}).

\begin{figure}[htbp]
\centering
\includegraphics{graphics/fig_ch2_2.pdf}
\caption{So sánh luồng RAG tìm một lần với vòng lặp agent gọi công cụ qua MCP}

Khác với Toolformer hay Self-RAG, đề tài không huấn luyện thêm mô hình; hành vi của agent được điều khiển bằng lời nhắc và mô tả công cụ, nên đóng góp nằm ở thiết kế công cụ cho miền pháp luật.

MCP \cite{anthropic2024mcp} là giao thức mở do Anthropic công bố tháng 11/2024: máy chủ (server) khai báo công cụ với tên, mô tả và lược đồ tham số; ứng dụng (client) đưa mô tả cho mô hình và chuyển lời gọi của mô hình tới máy chủ, nên một máy chủ phục vụ được nhiều client. Trong đề tài, máy chủ MCP cung cấp bốn công cụ chỉ đọc \texttt{search}, \texttt{hierarchical\_navigate}, \texttt{graph\_traverse}, \texttt{get\_raw}, và client là Claude Code chạy không tương tác.

\subsection{Các độ đo đánh giá}
\label{subsec:c2_do_do}

\textbf{Hit@k và MRR.} Gọi $Q$ là tập câu hỏi và $\mathrm{rank}_q$ là thứ hạng của điều khoản đáp án đầu tiên cho câu hỏi $q$ ($1/\infty = 0$ nếu không tìm thấy); một kết quả được coi là trúng khi nó là điều khoản đáp án hoặc nằm bên dưới điều khoản đáp án trong cây:
\begin{equation}
\label{eq:c2_mrr}
\mathrm{Hit@}k = \frac{1}{|Q|} \sum_{q \in Q} \mathbf{1}\left[\mathrm{rank}_q \le k\right], \qquad
\mathrm{MRR} = \frac{1}{|Q|} \sum_{q \in Q} \frac{1}{\mathrm{rank}_q}.
\end{equation}

\textbf{Kiểm định McNemar.} Khi so hai cấu hình A và B trên cùng tập câu hỏi, gọi $b$ là số câu A đúng còn B sai và $c$ là số câu A sai còn B đúng. Kiểm định McNemar chính xác \cite{mcnemar1947} kiểm tra giả thuyết hai loại bất đồng có cùng xác suất, với $n = b + c$:
\begin{equation}
\label{eq:c2_mcnemar}
p = \min\left(1,\; 2 \sum_{i=0}^{\min(b, c)} \binom{n}{i} \left(\frac{1}{2}\right)^{n}\right).
\end{equation}

\textbf{Khoảng tin cậy Wilson.} Với tỉ lệ quan sát $\hat{p}$ trên $n$ mẫu và $z = 1{,}96$ (mức tin cậy 95\%), khoảng Wilson luôn nằm trong $[0, 1]$ và vẫn đáng tin khi $\hat{p}$ gần 0 hoặc 1 với mẫu nhỏ, như ở bộ đánh giá câu trả lời:
\begin{equation}
\label{eq:c2_wilson}
\frac{\hat{p} + \dfrac{z^2}{2n} \pm z\sqrt{\dfrac{\hat{p}(1 - \hat{p})}{n} + \dfrac{z^2}{4n^2}}}{1 + \dfrac{z^2}{n}}.
\end{equation}

\textbf{Đánh giá câu trả lời.} RAGAS \cite{es2024ragas} và ARES \cite{saadfalcon2024ares} dùng mô hình ngôn ngữ để chấm độ trung thành và độ liên quan; ALCE \cite{gao2023alce} đánh giá chất lượng trích dẫn. Đề tài chấm bằng quy tắc dựa trên đáp án tra cứu thủ công (Mục~\ref{sec:ch5-cau-tra-loi}); tiêu chí ``mọi số hiệu Điều và số tiền trong câu trả lời đều có trong nguồn được trích'' nhắm vào loại lỗi nguy hiểm nhất mà \cite{magesh2025hallucination} chỉ ra: trích dẫn hoặc con số nghe hợp lý nhưng sai.
